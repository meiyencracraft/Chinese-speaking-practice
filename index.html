        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";

        const firebaseConfig = JSON.parse(typeof __firebase_config !== 'undefined' ? __firebase_config : '{}');

        const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app';

        

        let db;

        let auth;

        let user;


        if (Object.keys(firebaseConfig).length > 0) {

            const app = initializeApp(firebaseConfig);

            auth = getAuth(app);

            db = getFirestore(app);

            

            const initAuth = async () => {

                try {

                    if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {

                        const userCredential = await signInWithCustomToken(auth, __initial_auth_token);

                        user = userCredential.user;

                        console.log("Firebase Auth Success (Custom Token)");

                    } else {

                        const userCredential = await signInAnonymously(auth);

                        user = userCredential.user;

                        console.log("Firebase Auth Success (Anonymous)");

                    }

                } catch (error) {

                    console.error("Auth Error", error);

                    try {

                         const userCredential = await signInAnonymously(auth);

                         user = userCredential.user;

                    } catch(e) {

                         console.error("Auth Fallback Error", e);

                    }

                }

            };

            initAuth();

        }


        window.uploadScoreData = async (studentName, sessionData) => {

            if (!db || !user) {

                window.showCustomAlert("数据库连接未就绪 / Database connection not ready", "提示 / Notice");

                return;

            }


            try {

                const colRef = collection(db, 'artifacts', appId, 'public', 'data', 'tone_practice_scores_v6');

                

                let totalAttempts = 0;

                let totalScore = 0;

                let attemptedCount = 0;

                

                const details = Object.values(sessionData).map(d => {

                    totalAttempts += d.attempts;

                    totalScore += d.bestScore;

                    if (d.attempts > 0) attemptedCount++;

                    return d;

                });


                const totalItems = Object.keys(sessionData).length;

                const avgScore = attemptedCount > 0 ? Math.round(totalScore / attemptedCount) : 0;


                await addDoc(colRef, {

                    studentName: studentName,

                    timestamp: serverTimestamp(),

                    dateStr: new Date().toLocaleString('zh-CN'),

                    avgScore: avgScore,

                    totalAttempts: totalAttempts,

                    attemptedCount: attemptedCount,

                    totalItems: totalItems,

                    details: JSON.stringify(details)

                });


                window.showCustomAlert("成绩提交成功！数据已永久保存 / Scores successfully submitted!", "成功 / Success");

                

            } catch (e) {

                console.error("Upload failed: ", e);

                window.showCustomAlert("提交失败: " + e.message + " / Submission failed", "错误 / Error");

            }

        };


        window.fetchTeacherData = async () => {

             if (!db || !user) return [];

             try {

                const colRef = collection(db, 'artifacts', appId, 'public', 'data', 'tone_practice_scores_v6');

                const snapshot = await getDocs(colRef);

                let data = [];

                snapshot.forEach(documentSnapshot => {

                    data.push({

                        id: documentSnapshot.id,

                        ...documentSnapshot.data()

                    });

                });

                data.sort((a, b) => (b.dateStr || '').localeCompare(a.dateStr || ''));

                return data;

             } catch (e) {

                 console.error("Fetch failed", e);

                 window.showCustomAlert("获取数据失败 / Failed to fetch data", "错误 / Error");

                 return [];

             }

        };


        window.clearAllTeacherDataFromCloud = async () => {

             if (!db || !user) return false;

             try {

                const colRef = collection(db, 'artifacts', appId, 'public', 'data', 'tone_practice_scores_v6');

                const snapshot = await getDocs(colRef);

                const deletePromises = [];

                snapshot.forEach(documentSnapshot => {

                    deletePromises.push(deleteDoc(doc(db, 'artifacts', appId, 'public', 'data', 'tone_practice_scores_v6', documentSnapshot.id)));

                });

                await Promise.all(deletePromises);

                return true;

             } catch (e) {

                 console.error("Clear failed", e);

                 return false;

             }

        };

            recognition.onresult = (event) => {

                let currentFinal = '';

                    } else {

                    }

                }

                const display = currentFinal + currentInterim;

                if(tElem && display) {

                    const spokenPy = getSpokenPinyin(display);

                    tElem.innerText = `${display} ${spokenPy}`;

                } else if(tElem) {

        function calculateAndScore(uniqueId, userSpeech, duration) {

            const target = sessionData[uniqueId].text;

            if (!userSpeech || userSpeech.trim().length === 0) return;

            const similarity = calculateSimilarity(target, userSpeech);

            let rawScore = Math.round(similarity * 100);

            

            const modeAtRecording = globalStudentStrictnessMode;

            let score = rawScore;

            if (modeAtRecording === 'encouraging') {

                score = Math.min(100, Math.round(rawScore * 1.15 + 10));

                if (rawScore > 40 && score < 75) score = 75;

            } else if (modeAtRecording === 'strict') {

                score = Math.max(0, Math.round(rawScore * 0.9 - 5));

                if (rawScore < 95) score = Math.min(score, rawScore);

            }


            const spokenPy = getSpokenPinyin(userSpeech);


            sessionData[uniqueId].attempts += 1;

            sessionData[uniqueId].lastDuration = duration;

            sessionData[uniqueId].strictnessMode = modeAtRecording;

            sessionData[uniqueId].spokenTranscript = userSpeech;

            sessionData[uniqueId].spokenPinyin = spokenPy;

            

            if (score > sessionData[uniqueId].bestScore) {

                sessionData[uniqueId].bestScore = score;

            }


            const scoreDisplay = document.getElementById(`score-display-${uniqueId}`);

            let colorClass = score >= 80 ? 'text-green-500' : (score >= 60 ? 'text-yellow-500' : 'text-red-500');

            scoreDisplay.className = `text-2xl font-bold ${colorClass} whitespace-nowrap pt-2 font-sans`;

            scoreDisplay.innerText = score + "分 / pts";


            const tElem = document.getElementById(`transcript-${uniqueId}`);

            if (tElem) {

                tElem.innerText = `${userSpeech} ${spokenPy}`;

            }


            const card = document.getElementById(`card-${uniqueId}`);

            if (score >= 80) {

                card.classList.add('border-green-200', 'bg-green-50');

                setTimeout(() => card.classList.remove('border-green-200', 'bg-green-50'), 2000);

            }

        }

        function calculateSimilarity(s1, s2) {

            const cleanS1 = s1.replace(/[^\w\u4e00-\u9fa5]/g, "");

            const cleanS2 = s2.replace(/[^\w\u4e00-\u9fa5]/g, "");

            if (cleanS1.length === 0 || cleanS2.length === 0) return 0;

            const longer = cleanS1.length > cleanS2.length ? cleanS1 : cleanS2;

            const shorter = cleanS1.length > cleanS2.length ? cleanS2 : cleanS1;

            const longerLength = longer.length;

            if (longerLength === 0) return 1.0;

            return (longerLength - levenshteinDistance(longer, shorter)) / longerLength;

        }


        function levenshteinDistance(s, t) {

            if (!s.length) return t.length;

            if (!t.length) return s.length;

            const arr = [];

            for (let i = 0; i <= t.length; i++) {

                arr[i] = [i];

                for (let j = 1; j <= s.length; j++) {

                    arr[i][j] = i === 0 ? j : Math.min(arr[i-1][j]+1, arr[i][j-1]+1, arr[i-1][j-1]+(s[j-1]===t[i-1]?0:1));

                }

            }

            return arr[t.length][s.length];

        }
